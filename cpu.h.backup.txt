#pragma once
#include <cstdint>
#include "Bus.h"
#include <string>
#include <functional>

// http://www.codeslinger.co.uk/pages/projects/gameboy/files/GB.pdf

class CPU
{
public:
	CPU();
	~CPU();

	void connectBus(Bus* b) { bus = b; }

public:

	// Flag Register
	union FlagRegister {
		struct {
			uint8_t Z : 1; // This bit is set when the result of a math operation is zero or two values match when using the CP instruction.
			uint8_t N : 1; // This bit is set if a subtraction was performed in the last math instruction.
			uint8_t H : 1; // This bit is set if a carry occurred from the lower nibble in the last math operation.
			uint8_t C : 1; // This bit is set if a carry occurred from the last math operation or if register A is the smaller value when executing the CP instruction
			uint8_t unused : 4;

		};
		uint8_t reg;
	};

	// Registers
	union AF {
		struct {
			uint8_t A;
			FlagRegister F;
		};
		uint16_t reg;
	};
	union BC {
		struct {
			uint8_t B;
			uint8_t C;
		};
		uint16_t reg;
	};
	union DE {
		struct {
			uint8_t D;
			uint8_t E;
		};
		uint16_t reg;
	};
	union HL {
		struct {
			uint8_t H;
			uint8_t L;
		};
		uint16_t reg;
	};

	uint16_t SP, PC; // Stack and Program Counter
	AF AF;
	BC BC;
	DE DE;
	HL HL;

	// Opcodes
	/*
	void LD_INM(uint8_t* r1, uint8_t immediate);
	void LD_R1R2(uint8_t* r1, uint8_t* r2);
	void LD_A(uint8_t data);
	void LD_n_with_A(uint8_t* n); // n = A,B,C,D,E,H,L,(BC),(DE),(HL),  [(nn) = two byte immediate value. (LS byte first.)]
	void LD_AC(); // Put value at address $FF00 + register C into A. Same as : LD A, ($FF00 + C)
	void LD_CA(); // Put A into address $FF00 + register C.
	void LDD_A_HL(); // Put value at address HL into A. Decrement HL.	Same as: LD A,(HL) - DEC HL
	void LDD_HL_A(); // Put A into memory address HL. Decrement HL.		Same as: LD (HL),A - DEC HL
	void LDI_A_HL(); // Put value at address HL into A. Increment HL.	Same as: LD A,(HL) - INC HL
	void LDI_HL_A(); // Put A into memory address HL. Increment HL.		Same as: LD (HL),A - INC HL
	*/

	// 8-bit load
	void LD(uint8_t* dest, uint8_t value);

	// 16-bit load
	void LD(uint16_t* dest, uint16_t value);
	void PUSH(uint16_t value);
	void POP(uint16_t* dest);

	// 8-bit ALU
	void ADD(uint8_t value); // Add value to A
	void ADC(uint8_t value); // Add value + Carry flag to A
	void SUB(uint8_t value); // Subtract value from A
	void SBC(uint8_t value); // Subtract value + Carry flag from A.
	void AND(uint8_t value); // Logically AND value with A, result in A.
	void OR(uint8_t value); // Logical OR value with register A, result in A.
	void XOR(uint8_t value); // Logical exclusive OR value with register A, result in A.
	void CP(uint8_t value); // Compare A with value. This is basically an A - value subtraction instruction but the results are thrown away.
	void INC(uint8_t* n); // Increment register n.
	void DEC(uint8_t* n); // Decrement register n.

	// 16-bit Arithmetic
	void ADD(uint16_t value); // Add value to HL.
	void ADD(int8_t signed_value); // Add one byte signed immediate value to SP.
	void INC(uint16_t* nn); // Increment register nn.
	void DEC(uint16_t* nn); // Decrement register nn.

	// Miscellaneous
	void SWAP(uint8_t* n); // Swap upper & lower nibles of n.
	void DDA(); // Decimal adjust register A. This instruction adjusts register A so that the correct representation of Binary Coded Decimal(BCD) is obtained.
	void CPL(); // Complement A register. (Flip all bits.)
	void CCF(); // Complement carry flag. If C flag is set, then reset it. If C flag is reset, then set it.
	void SCF(); // Set Carry flag.
	void NOP(); // No operation.
	void HALT(); // Power down CPU until an interrupt occurs. Use this when ever possible to reduce energy consumption.
	void STOP(); // Halt CPU & LCD display until button pressed.
	void DI(); // This instruction disables interrupts but not immediately. Interrupts are disabled after instruction after DI is executed.
	void EI(); // This instruction disables interrupts but not immediately. Interrupts are disabled after instruction after DI is executed.

	// Rotates & Shifts
	void RLCA(); // Rotate A left. Old bit 7 to Carry flag.
	void RLA(); // Rotate A left through Carry flag.
	void RRCA(); // Rotate A right. Old bit 0 to Carry flag.
	void RRA(); // Rotate A right through Carry flag.
	void RLC(uint8_t* n); // Rotate n left. Old bit 7 to Carry flag.
	void RL(uint8_t* n); // Rotate n left through Carry flag.
	void RRC(uint8_t* n); // Rotate n right. Old bit 0 to Carry flag.
	void RR(uint8_t* n); // Rotate n right through Carry flag.
	void SLA(uint8_t* n); // Shift n left into Carry. LSB of n set to 0.
	void SRA(uint8_t* n); // Shift n right into Carry. MSB doesn't change.
	void SRL(uint8_t* n); // Shift n right into Carry. MSB set to 0.

	// Bit Opcodes
	void BIT(uint8_t bit, uint8_t* r); // Test bit b in register r.
	void SET(uint8_t bit, uint8_t* r); // Set bit b in register r.
	void RES(uint8_t bit, uint8_t* r); // Reset bit b in register r.

	// Jumps
	void JP(uint16_t addr); // Jump to address nn. two byte immediate value. (LS byte first.)
	void JP(uint8_t cc, uint16_t addr); // Jump to address n if cc condition is true.
	void JR(uint8_t addr); // Add n to current address and jump to it.
	void JR(uint8_t cc, uint8_t addr); // Jump to address n if cc condition is true.

	// Calls
	void CALL(uint16_t nn); // Push address of next instruction onto stack and then jump to address nn. two byte immediate value. (LS byte first.)
	void CALL(uint8_t cc, uint16_t nn); // Call address nn if following condition is true. two byte immediate value. (LS byte first.)

	// Restarts
	void RST(uint8_t n); // Push present address onto stack. Jump to address $0000 + n. n = $00,$08,$10,$18,$20,$28,$30,$38

	// Returns
	void RET(); // Pop two bytes from stack & jump to that address.
	void RET(uint8_t cc); // Return if cc condition is true.
	void RETI(); // Pop two bytes from stack & jump to that address then enable interrupts.




	void clock();

	uint8_t opcode = 0x00;

private:
	Bus* bus = nullptr;
	void write(uint16_t addr, uint8_t data);
	uint8_t read(uint16_t addr);

	struct INSTRUCTION {
		std::string name;
		std::function<void()> function;
	};

	std::array<INSTRUCTION, 2> lookup = { {"NOP", } };
};
